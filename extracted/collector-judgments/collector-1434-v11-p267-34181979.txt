‫‪267‬‬
‫ادعــى املدعــي أن املدعــى عليهــا كانــت زوجتــه وأجنــب منهــا بنــت‬
‫وعمرهــا اآلن اثنــي عشــرة ســنة ومــن ذلــك الوقــت هــي يف حضانــة‬
‫املدعــى عليهــا أمهــا وقــد تزوجــت املدعــى عليهــا قبــل خمــس ســنني‬
‫ويطلــب احلكــم لــه بحضانــة بنتــه املذكــورة هكــذا ادعــى‪ ،‬بعــرض‬
‫دعــوى املدعــي علــى املدعــى عليهــا املعــرف بهــا مــن قبــل والدهــا صادقــت‬
‫علــى مــا ذكــره املدعــي وذكــرت أن الطــاق كان قبــل إحــدى عشــرة‬
‫ســنة وقــد تزوجــت بعــده بثمــان ســنوات تقريب ـاً‪ ،‬ومنــذ زواجهــا والبنــت‬
‫متكــث عنــد والدهــا املدعــي وبعــد ثــاث ســنوات مــن زواجهــا أخــذت‬
‫البنــت عندهــا برضــا املدعــي ألنهــا أصيبــت مبــرض الســكري نتيجــة‬
‫لقلــة الرعايــة واالهتمــام وعــدم تواجــد والدهــا يف املنطقــة الرتباطــه‬
‫بالعمــل العســكري يف شــرورة حيــث إنهــا متكــث أحيان ـ ًا عنــد زوجــة‬
‫والدهــا ومــرة أخــرى عنــد جدتهــا ألبيهــا وكل هــؤالء ال جتــد عندهــم‬
‫الرعايــة واالهتمــام الالزمــن ملثلهــا وال توافــق علــى مــا طلــب بــه املدعــي‬
‫واحلكــم لهــا بحضانتهــا هكــذا أجابــت‪ ،‬بعــرض ذلــك علــى املدعــي‬
‫أجــاب مــا ذكرتــه املدعــى عليهــا غيــر صحيــح فالبنــت جتــد عنــده‬
‫الرعايــة واالهتمــام علــى أحســن مــا يكــون وطلــب حضانتهــا‪ ،‬جــرى‬
‫محاولــة اإلصــاح بــن الطرفــن‪ ،‬حيــث إن احلضانــة شــرعت ملصلحــة‬
‫احملضــون‪ ،‬وهــذا يختلــف باختــاف األحــوال واألشــخاص وحيــث إن‬
‫الولــد املتنــازع عليــه بــن الطرفــن أنثــى وحاجتهــا إلــى أمهــا وبقاؤهــا‬
‫عندهــا أنفــع لهــا وبــه يتحقــق مــن مصاحلها ما ال ميكــن حتققه وهي عند‬
‫والدهــا ال ســيما وأن والدهــا يعمــل عســكري ًا ويلحقــه مــن االســتالمات‬

      ‫واملناوبــات ممــا يجعلــه بعيــد ًا عــن بنتــه‪ ،‬وحيــث إن مــدة اســتالمه قــد‬
      ‫تصــل إلــى خمســة عشــر يومـاً‪ ،‬ويف ذلــك تعطيــل ملصالــح البنــت وبقائهــا‬
      ‫عنــد أمهــا أحفــظ لهــا واصــون حلاجتهــا إلــى تعلــم مــا ينفعهــا يف دينهــا‬
      ‫ودنياهــا ممــا ينبغــي علــى النســاء معرفتــه كالقيــام مبصالــح البيــت‬
      ‫ونحــو ذلــك ممــا هــي يف حاجــة ماســة إليــه‪ ،‬ولكــون األم أشــفق وأنصــح‬
      ‫لبنتهــا مــن زوجــة أبيهــا ومصلحتهــا بقاؤهــا عنــد أمهــا‪ ،‬صــدر احلكــم‬
      ‫بصــرف النظــر عــن دعــوى املدعــى وجعــل احلضانــة لــأم ‪ .‬قــرر املدعــى‬
          ‫عــدم قناعتــه باحلكــم‪ ،‬صــدق احلكــم مــن محكمــة االســتئناف ‪.‬‬




      ‫احلمد هلل وحده وبعد فلدي أنا ‪ ...‬القاضي يف احملكمة العامة مبحافظة‬
      ‫قلــوة وبنــاء علــى املعاملــة احملالــة لنــا مــن فضيلــة رئيــس احملكمــة‬
      ‫العامــة مبحافظــة قلــوة برقــم ‪ 3475176‬وتاريــخ ‪1434/02/16‬ه ـ‬

      ‫يــوم الثالثــاء املوافــق‪ 1434/04/02‬ه ـ افتتحــت اجللســة الســاعة احلاديــة‬

      ‫رقــم ‪ ...‬وحضــر حلضــوره املدعــى عليهــا‪ ... ...‬ســعودية بالســجل املدنــي‬
      ‫رقــم ‪ ...‬واملعــرف بهــا مــن قبــل والدهــا احلاضــر معهــا ‪ ...‬ســعودي بالســجل‬
      ‫املدنــي ‪ ...‬وبســؤال املدعــي عــن دعــواه ادعــى قائـاً‪ :‬لقــد كانــت املدعــى‬
      ‫عليهــا زوجتــي وأجنبــت منهــا بنت اســمها ‪ ...‬وعمرها اآلن اثنا عشــر ســنة‬
      ‫ومنــذ ذلــك الوقــت وبنتــي‪ ...‬يف حضانتهــا وقــد تزوجت املدعــى عليها قبل‬
      ‫خمــس ســنوات تقريبـ ًا أطلــب احلكــم لــي بحضانــة بنتــي املذكــورة هــذه‬
      ‫دعــواي وبعــرض الدعــوى علــى املدعــى عليهــا أجابــت قائلــة ‪ :‬مــا ذكــره‬

‫‪269‬‬
‫املدعــي يف دعــواه أنــه كان زوج ـ ًا لــي وأجنبــت منــه البنــت املذكــورة‬
‫وطلقنــي قبــل أحــد عشــر ســنة صحيــح وقــد تزوجــت بعــده قبــل ثمــان‬
‫ســنوات تقريبـ ًا ومنــذ زواجــي والبنــت متكــث عنــد والدهــا املدعــي وبعــد‬
‫ثــاث ســنوات مــن زواجــي أخــذت بنتــي ‪ ...‬عنــدي برضــاه ألنهــا اصيبــت‬
‫مبــرض الســكري نتيجــة لقلــة الرعايــة واإلهتمــام وعــدم تواجــد والدهــا‬
‫يف املنطقــة الرتباطــه بالعمــل العســكري يف شــرورة حيــث أنهــا متكــث‬
‫احيانـ ًا عنــد زوجــة والدهــا ومــرة آخــرى عنــد جدتهــا ألبيهــا وكل هــؤالء‬
‫ال جتــد عندهــم الرعايــة واالهتمــام الالزمــن ملثلهــا وال اوافــق علــى مــا‬
‫طالــب بــه املدعــي واحلكــم لــه بحضانتهــا هكــذا أجابــت وبعــرض ذلــك‬
‫ا ماذكــرت املدعــى عليهــا غيــر صحيــح فبنتــي‬     ‫علــى املدعــي أجــاب قائـ ً‬
‫‪ ...‬جتــد الرعايــة واالهتمــام عنــدي علــى أحســن مــا يكــون وأنــا أطلــب‬
‫بحضانتهــا لــي هكــذا قــرر وبعــرض مــا ذكــر املدعــي علــى املدعــى‬
‫عليهــا قالــت ماذكــره املدعــي غيــر صحيــح وقــد ســبق وأن كلمتنــي‬
‫زوجتــه وقالــت لــي أنــا مانــي مكلفــه ببنتــك هكــذا قــررت ثــم جــرى‬
‫ســؤال املدعــي عــن مقــر ســكنه وعملــه قــال أنــا اســكن يف مكــة‬
‫املكرمــة وأعمــل عســكري يف حــرس احلــدود يف شــرورة هكــذا‬
‫قــرر وبســؤاله كيــف طريقــة إســتالمه يف عملــه قــال امكــث خمســة‬
‫عشــر يومـ ًا يف شــرورة وخســمة عشــر يومـ ًا يف مكــة واحيانـ ُا اســبوع يف‬
‫شــرورة واســبوع يف مكــة هــذا قــرر وبســؤاله عــن الفتــرة التــي يكــون‬
‫فيهــا يف شــرورة أيــن تكــون بنتــه قــال تكــون يف هــذه الفتــرة يف بيتــي‬
‫عنــد زوجتــي وأحيانـ ًا عنــد أمــي هكــذا قــرر وبســؤال املدعــى عليهــا عــن‬
‫عملهــا وســكنها قالــت أســكن يف قلــوة وأعمــل معلمــة مديــرة مدرســة‬
‫هكــذا قــررت ثــم جــرى منــي محاولــة االصــاح بــن الطرفــن إال انــه‬
