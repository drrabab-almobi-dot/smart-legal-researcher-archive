       ‫عقــار ‪ -‬التملــك الصــوري للعقــار ‪ -‬صنــدوق التنميــة العقــاري ‪ -‬اتفــاق‬
       ‫على ســداد أقســاط قرض ‪ -‬ســداد أجرة ســكن عائلي مشــترك ‪ -‬صورية‬
       ‫متلــك العقــار ‪ -‬شــهادة غيــر منتجــه يف الدعــوى ‪ -‬ضــم صــورة صــك‬
       ‫ســابق إلــى الدعــوى القائمــة ‪ -‬حتــول العقــد الصــوري إلــى حقيقــي ‪-‬‬
                                                       ‫صــرف النظــر عــن الدعــوى‬




       ‫مــا أســتند إليــه القاضــي مــن املبــادئ العامــة وقواعد العدالة يف تســبيب‬
                                                                             ‫حكمه‬



       ‫تقــدم املدعــي بدعــوى ضــد املدعــى عليــه «شــقيقه» بــأن مــن اجلــاري يف‬
       ‫ملكــه البيــت املســجل باســمه مبوجــب صــك شــرعي وقــد نــال ألجلــه‬
       ‫قرضــا مــن صنــدوق التنميــة العقــاري وكان يعمــل يف مدينــة أخــرى‬
       ‫ووالــده وبعــض أخوتــه يســكنون يف البيــت محــل النــزاع ومــن بينهــم‬
       ‫املدعــى عليــه ويف عــام ‪1410‬ه ـ اتفــق خمســة مــن اإلخــوة ومعهــم املدعــى‬
       ‫عليــه علــى أن يقومــوا باملشــاركة يف دفــع جــزء مــن قســط القــرض‬
       ‫الســنوي ويكــون علــى كل واحــد منهــم ألفــان وأربعمئــة ريــال يدفعهــا‬
       ‫كل ســنة وألن املدعــى عليــه بقــي يف العقــار أحــدى عشــرة ســنة فيطلــب‬
       ‫إلزامــه بدفــع أربعــة وعشــرين ألــف ريــال ‪ ،‬أنكــر املدعــى عليــه الدعــوى‬
‫‪309‬‬
‫ودفــع بــأن املنــزل ملــك لوالــده حقيقــة آلــت أرضــه إليــه مبنحــة مــن األمانة‬
‫وألنــه ســبق وأن اقتــرض مــن صنــدوق التنميــة فقــد طلــب مــن املدعــي أن‬
‫يتقــدم بطلــب قــرض عليهــا فجــرى تســجيل األرض باســمه صوريــا وإال‬
‫فــإن مــن بنــاه والــده وقــد دفــع قســطني وأعفــي مــن قســطني ولــو طلــب‬
‫والــده منــه دفــع أي مبلــغ لوافــق ويف أزمــة اخلليــج وملــا كثــرت العوائــل يف‬
‫البيــت طلــب والــده مــن اإلخــوة أن يرســل كل منهــم ألفــن وخمســمئة‬
‫ريــال شــهريا فاجتمــع لديــه مبلغــا وطلــب املدعــي الــزواج فدفــع لــه‬
‫ســتني ألــف ريــال ثــم طلــب مــن والــده تســليمه الصــك فلمــا ســلمه لــم‬
‫يــرده إليــه حتــى اآلن حتــى قــام والــده بإقامــة دعــوى ضــده إلثبــات متلــك‬
‫البيــت ‪ -‬حضــر والــد املتداعيــن وقــرر بــأن مــا ورد يف دعــوى املدعــي‬
‫غيــر صحيــح وأن الصحيــح بــأن أوالده قامــوا بإعطائــه مــاال أيــام أزمــة‬
‫اخلليــج عــدا املدعــي فلــم يســلمه شــيئا بــل وهبــه ســتني ألــف ريــال لزواجــه‬
‫وأنــه مــن قــام ببنــاء البيــت بــا مشــاركة مــن املدعــي ‪ -‬أحضــر املدعــي‬
‫شــاهدا أحــد إخوتــه وشــهد باتفــاق اإلخــوة علــى أن يدفــع كل واحــد‬
‫منهــم ألفــن وأربعمئــة ريــال لســداد القــرض عــن املدعــي ‪ -‬بعــرض‬
‫الشــهادة علــى املدعــى عليــه أنكرهــا ‪ -‬جــرى اطــاع الدائــرة علــى‬
‫صــورة صــك املتضمــن دعــوى والــد املدعــي ضــده وقــد أقــر فيــه صراحــة‬
‫بــأن اإلفــراغ لــه كان صوريــا وأحيلــت املعاملــة ملكتــب الصلــح وتضمــن‬
‫ثبــوت الصلــح بينهمــا علــى إبقــاء العقــار مســجال باســم املدعــي مقابــل‬
‫مئــة ألــف ريــال ‪ -‬عليــه وحيــث حضــر والــد املتداعيــن وأفــاد أن أصــل‬
‫متلــك البيــت احلقيقــي لــه ومتلــك املدعــي للعقــار متلــكا صوريــا وصــادق‬
‫املدعــي علــى ذلــك بــل اصطلــح مــع أبيــه علــى أن يدفــع لــه مئــة ألــف ريــال‬
‫ودفعهــا لــذا لــم حتصــل ملكيــة املدعــي للبيــت متلــكا حقيقيــا إال بعــد‬

      ‫الصلــح وألن مطالبــة املدعــي باســتحقاق املبلــغ املدعــى بــه قبــل متلكــه‬
      ‫للبيــت متلــكا حقيقــا لذلــك كلــه فقــد قــررت الدائــرة صــرف النظــر‬
      ‫عــن دعــوى املدعــي وأخلــت ســبيل املدعــى عليــه مــن هــذه القضيــة ‪ -‬قــرر‬
      ‫املدعــي االعتــراض فأفهــم بالتعليمــات ‪ -‬صــدق احلكــم مــن محكمــة‬

      ‫قائــا يف دعــواه إن مــن اجلــاري يف ملكــي وحتــت تصــريف كامــل البيــت‬
      ‫الفيــا مكونــة مــن دوريــن مســتقلني يقــع يف حــي الســعودية يف اخلــرج‬
      ‫اململوكــة لــي بالصــك الصــادر مــن كتابــة عــدل اخلــرج برقــم ‪1/230‬‬
      ‫يف ‪1403/4/11‬ه ـ وقــد اقترضــت عليهــا مــن صنــدوق التنميــة العقاريــة‬
      ‫مبلغــا قــدره ثالثمائــة ألــف ريــال وكنــت أعمــل يف املنطقــة الشــرقية‬
      ‫وقــد رغــب والــدي ســكن الفيــا وطلــب مــن أخوتــي الســكن معــه ومــن‬
      ‫ضمنهــم املدعــى عليــه ويف عــام ‪1410‬ه ـ اتفقنــا نحــن األخــوة اخلمســة‬

      ‫القســط الســنوي لصنــدوق التنميــة العقاريــة البالــغ اثنــا عشــر ألــف ريــال‬

‫‪311‬‬
‫علينــا نحــن اخلمســة فبلــغ نصيــب الواحــد ألفــان وأربعمائــة ريــال يدفعهــا‬
‫كل شــخص عــن كل ســنة وقــد بقــي املدعــى عليــه يف الســكن مــن‬
‫عــام ‪1410‬ه ـ إلــى عــام ‪1421‬ه ـ وبلــغ إجمالــي ماعليــه عشــر ســنني أي‬
‫أربعــة وعشــرون ألــف ريــال وصنــدوق التنميــة العقاريــة الزال يطالبنــي‬
‫أطلــب احلكــم علــى املدعــى عليــه وإلزامــه بــأن يدفــع هــذا املبلــغ لــي‬
‫هــذه دعــواي وبســؤال املدعــى عليــه قــال ال صحــة جلميــع ماذكــره‬
‫املدعــي جملــة وتفصيــا وذلــك أن البيــت يف األســاس لوالدنــا وأرضــه‬
‫منحــه عــن طريــق بلديــة اخلــرج كغيرهــا مــن أراضــي حــي الســعودية‬
‫وســبق للوالــد أن أقتــرض مــن صنــدوق التنميــة العقاريــة يف املدينــة (‬
‫محافظــة خيبــر) وألن املدعــي يف ذلــك الوقــت يحمــل بطاقــة األحــوال‬
‫جعــل والدنــا البيــت بإســم املدعــي صوريــا ووالــدي هــو الــذي يدفــع‬
‫قــرض صنــدوق التنميــة ودفــع قســطني وأعفــي مــن قســطني وهــو الــذي‬
‫يتصــرف يف البيــت ولــو طلــب منــا دفــع أي مبلــغ دفعنــاه وقــد اجتمعــت‬
‫عوائلنــا يف البيــت عــام ‪1410‬ه ـ يف بدايــة أزمــة اخلليــج األولــى ولــم أكــن‬
‫وقتهــا متزوج ـ ًا وحــن كثــرت العائلــة يف املنــزل حيــث أنــا أربعــة أخــوة‬
‫يف اجلبهــة وعوائــل أخوانــي عنــد والــدي طلــب منــا الوالــد أن يرســل كل‬
‫واحــد منــا شــهريا ألفــن وخمســمائة ريــال واجتمــع عنــد والدنــا مبلغــا‬
‫وقــد طلــب املدعــي الــزواج فدفــع عنــه الوالــد ســتني ألــف ريــال ثــم أن‬
‫املدعــي طلــب مــن والــدي أن يســلم صــك البيــت مــن أجــل أن يســقط‬
‫قــرض الصنــدوق فســلمه لــه ولــم يعــده لوالــدي حتــى أقــام والــدي دعــوى‬
‫ضــد املدعــي وذلــك مــن أجــل إثبــات متلكــه للبيــت متلــك ًا حقيقيــا حيــث‬
